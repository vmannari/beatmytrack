class Backend extends Node:
	signal identity_changed
	var player: Dictionary = {"id":"fixture-alice"}
	var enabled: bool = true
	var delayed: bool = false
	var calls: Array[String] = []
	var data: Dictionary = {}
	var data_by_player: Dictionary = {}
	func signed_in() -> bool: return enabled
	func base_url() -> String: return "http://fixture.invalid"
	func request_api(path: String, _method: int = 0, body: Dictionary = {}, _key: String = "", _auth: bool = false) -> Dictionary:
		calls.append(path)
		var result: Dictionary = data_by_player.get(str(player.get("id","")),data).duplicate(true)
		if path.ends_with("/featured"):
			data.states.race.featured = body.families.duplicate()
			result = {"featured":body.families.duplicate()}
		while delayed: await get_tree().process_frame
		return {"ok":true,"status":200,"data":result}

func _initialize() -> void: run.call_deferred()
func check(ok: bool, message: String) -> void:
	checks += 1
	if not ok:
		failures += 1
		print("FAIL: "+message)

func settle() -> void:
	await create_timer(0.45).timeout
	for tick: int in range(5): await process_frame
	if DisplayServer.get_name() != "headless": await RenderingServer.frame_post_draw

func capture(label: String) -> void:
	if DisplayServer.get_name() == "headless": return
	check(root.get_texture().get_image().save_png(output.path_join(label+".png")) == OK,"capture "+label)

func touch(position: Vector2, pressed: bool) -> void:
	var event := InputEventScreenTouch.new()
	event.index = 0
	event.position = position
	event.pressed = pressed
	root.push_input(event,true)

func swipe(position: Vector2, distance: float) -> void:
	touch(position,true)
	for step: int in range(1,5):
		var event := InputEventScreenDrag.new()
		event.index = 0
		event.position = position+Vector2(0,distance*step/4.0)
		event.relative = Vector2(0,distance/4.0)
		root.push_input(event,true)
		await process_frame
	touch(position+Vector2(0,distance),false)
	await settle()

func remote_state(metrics: Dictionary, scope: String) -> Dictionary:
	var data: Dictionary = {"metrics":metrics,"unlocked":{},"featured":[],"revision":1}
	for f: Dictionary in AchievementProgress.catalog().families:
		if f.scope != scope: continue
		for i: int in range(f.tiers.size()):
			if metrics.get(f.metric,0) >= f.tiers[i]: data.unlocked[f.id+"/"+str(i+1)] = 1792195200
	return data

func wall_point(showroom: CarShowroom, family: String, featured: bool = false) -> Vector2:
	var slot: Node3D = showroom.find_child(("FeaturedMedal_" if featured else "AchievementSlot_")+family,true,false)
	var point: Vector2 = showroom.camera.unproject_position(slot.global_position)*showroom.size/Vector2(showroom.showroom_viewport.size)
	return showroom.get_global_transform_with_canvas()*point
