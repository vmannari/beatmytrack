extends SceneTree

func _initialize() -> void: call_deferred("run")

func run() -> void:
	var key: String = "166f57396bcc750da426bf322777f9361f3815cbbf3c091202d7f10ede7e2573"
	var source: String = OS.get_environment("APPDATA").path_join("Circuit01Lab/cloud/"+key)
	var output: String = ProjectSettings.globalize_path("res://../build/analog-phone-deploy-2026-10-16")
	var snapshot: String = output.path_join("cache/"+key)
	var tracks: String = output.path_join("saved-tracks")
	var previous: Array = JSON.parse_string(FileAccess.get_file_as_string(ProjectSettings.globalize_path("res://../build/phone-drive-deploy-2026-10-14/source-tracks.json")))
	var saved_targets: Dictionary = {}
	for row: Dictionary in previous: saved_targets[str(row.file).get_basename()] = true
	DirAccess.make_dir_recursive_absolute(snapshot)
	DirAccess.make_dir_recursive_absolute(tracks)
	var index_bytes: PackedByteArray = FileAccess.get_file_as_bytes(source.path_join("index.json"))
	var index: Dictionary = JSON.parse_string(index_bytes.get_string_from_utf8())
	var manifest: Array[Dictionary] = []
	var seen: Dictionary = {}
	for entry: Dictionary in index.entries:
		var filename: String = str(entry.content_hash)+".json"
		var path: String = source.path_join(filename)
		assert(FileAccess.get_sha256(path) == entry.content_hash,"Published source hash mismatch")
		var bytes: PackedByteArray = FileAccess.get_file_as_bytes(path)
		var document: Dictionary = JSON.parse_string(bytes.get_string_from_utf8())
		assert(not seen.has(document.id),"Unique document identity")
		assert(TrackDocument.validate(document).is_empty(),"Valid current track schema")
		assert(document.has("start_line") and RoadPieces.route_from_pieces(document).ok,"Closed timed route and start line")
		var native: NativeTrack = TrackCompiler.compile(document)
		assert(native != null and native.route_length > 0,"Current native compilation")
		seen[document.id] = true
		var destinations: Array[String] = [snapshot.path_join(filename)]
		if saved_targets.has(document.id):
			destinations.append(tracks.path_join(str(document.id)+".json"))
			saved_targets.erase(document.id)
		for destination: String in destinations:
			var file := FileAccess.open(destination,FileAccess.WRITE)
			assert(file != null,"Open frozen track snapshot")
			file.store_buffer(bytes)
			file.close()
			assert(FileAccess.get_sha256(destination) == entry.content_hash,"Frozen track hash")
		manifest.append({"id":document.id,"name":document.name,"theme":document.theme,"hash":entry.content_hash,"route_length":native.route_length,"version_id":entry.version_id})
		print("PHONE_TRACK_VALIDATED: ",document.name)
	assert(manifest.size() == 27,"Full phone track catalog")
	assert(saved_targets.is_empty(),"All nine existing saved drafts matched")
	var index_file := FileAccess.open(snapshot.path_join("index.json"),FileAccess.WRITE)
	index_file.store_buffer(index_bytes)
	index_file.close()
	var report := FileAccess.open(output.path_join("validated-tracks.json"),FileAccess.WRITE)
	report.store_string(JSON.stringify({"count":manifest.size(),"cache_key":key,"tracks":manifest},"\t"))
	report.close()
	print("PHONE_TRACK_BUNDLE: ",manifest.size()," source hashes/schema/routes/native compilation/frozen copies passed")
	quit()
