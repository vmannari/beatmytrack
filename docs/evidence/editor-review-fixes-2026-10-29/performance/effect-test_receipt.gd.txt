extends SceneTree
## The last covered effect draw must include every prepared proxy and sentinel.
const EFFECTS = preload("res://vehicle/effect_render_preparation.gd")
var checks: int = 0
var failures: int = 0

func _initialize() -> void: run.call_deferred()
func check(ok: bool, label: String) -> void:
	checks += 1
	if not ok: failures += 1; print("FAIL: ",label)

func run() -> void:
	if DisplayServer.get_name() == "headless":
		print("GRAPHICAL_TEST_REQUIRED"); quit(2); return
	root.size = Vector2i(640,360)
	var owner := Node.new(); root.add_child(owner)
	var world := Node3D.new(); owner.add_child(world)
	var camera := Camera3D.new(); world.add_child(camera)
	camera.position = Vector3(0,20,0); camera.rotation.x = -PI/2
	camera.projection = Camera3D.PROJECTION_ORTHOGONAL; camera.size = 24; camera.current = true
	var sentinel := MeshInstance3D.new(); sentinel.mesh = BoxMesh.new(); world.add_child(sentinel)
	var effects := SandDust.new(); world.add_child(effects)
	var track: NativeTrack = TrackCompiler.compile(TrackDocument.create_default())
	effects.prepare_track(track)
	var expected: int = 0
	for child: Node in effects.get_children():
		if child is GPUParticles3D and child.is_visible_in_tree(): expected += 1
	check(expected > 0,"fixture has actual visible particle families")
	var observed: Dictionary = {"calls":0,"last_count":0,"sentinel":false}
	var draw: Callable = func() -> bool:
		observed.calls += 1
		var count: int = 0
		for node: Node in world.get_children():
			if node is GPUParticles3D and str(node.name).begins_with("Covered") and node.is_visible_in_tree(): count += 1
		observed.last_count = count
		observed.sentinel = sentinel.is_visible_in_tree()
		await RenderingServer.frame_post_draw
		return true
	var result: Dictionary = await EFFECTS.prepare(world,effects,camera,null,owner,draw,func() -> bool: return true)
	check(result.complete and bool(result.get("last_drawn",false)),"successful final actual draw supplies its completion receipt")
	check(result.particles == expected and observed.calls == expected and observed.last_count == expected,"last draw includes every configured particle proxy")
	check(observed.sentinel,"last draw retains the rest of the visible world")
	check(result.initialized.size() == expected,"every live particle proxy has its own submitted draw")
	EFFECTS.end(result)
	check(world.find_children("Covered*","GPUParticles3D",true,false).is_empty(),"finish restores proxy ownership without visible leftovers")
	var no_source_draws: int = observed.calls
	var empty: Dictionary = await EFFECTS.prepare(world,null,camera,null,owner,draw,func() -> bool: return true)
	check(empty.complete and not empty.last_drawn and observed.calls == no_source_draws,"no-particle preparation requests a separate covered draw")
	EFFECTS.end(empty)
	var gate: Dictionary = {"current":true,"draws":0}
	var cancelling: Callable = func() -> bool:
		gate.draws += 1
		await RenderingServer.frame_post_draw
		if gate.draws == 2: gate.current = false
		return true
	var interrupted: Dictionary = await EFFECTS.prepare(world,effects,camera,null,owner,cancelling,func() -> bool: return gate.current)
	check(not interrupted.complete and gate.draws == 2,"a stale scope stops even after an earlier draw")
	EFFECTS.end(interrupted)
	check(world.find_children("Covered*","GPUParticles3D",true,false).is_empty(),"cancellation removes all temporary proxies")
	var undrawn: Dictionary = await EFFECTS.prepare(world,effects,camera,null,owner,func() -> bool: return false,func() -> bool: return true)
	check(not undrawn.complete and not undrawn.last_drawn,"a missed frame cannot become a successful draw receipt")
	EFFECTS.end(undrawn)
	owner.queue_free(); await process_frame
	print("EFFECT_FINAL_DRAW_RECEIPT: %d checks, %d failures" % [checks,failures])
	quit(1 if failures else 0)
