extends SceneTree
## Matched native views plus construction checks for the owner's quality review.
const COACH = preload("res://vehicle/electric_coachwork.gd")
const MODEL = preload("res://vehicle/electric_car_model.gd")
var output: String = "res://../docs/evidence/electric-quality-pass-2026-10-25/native"
var checks: int = 0
var failures: int = 0
var metrics: Dictionary = {}

func _initialize() -> void: run.call_deferred()
func check(ok: bool, label: String) -> void:
	checks += 1
	if not ok:
		failures += 1
		print("FAIL: "+label)

func triangles(node: Node) -> int:
	var count: int = node.mesh.get_faces().size()/3 if node is MeshInstance3D else 0
	for child: Node in node.get_children(): count += triangles(child)
	return count

func settle() -> void:
	for frame: int in range(6): await process_frame
	await RenderingServer.frame_post_draw

func mesh_ray(car: CarVisual, prefix: String, origin: Vector3, direction: Vector3) -> Dictionary:
	var closest: float = INF
	var position_value := Vector3.ZERO
	for mesh: MeshInstance3D in car.body.find_children(prefix+"*","MeshInstance3D",true,false):
		var faces: PackedVector3Array = mesh.mesh.get_faces()
		for index: int in range(0,faces.size(),3):
			var point: Variant = Geometry3D.ray_intersects_triangle(origin,direction,faces[index],faces[index+1],faces[index+2])
			if point != null and origin.distance_squared_to(point)<closest:
				closest = origin.distance_squared_to(point)
				position_value = point
	return {"hit":is_finite(closest),"point":position_value}

func diagnose(car: CarVisual, camera: Camera3D, pixel: Vector2) -> Dictionary:
	var origin: Vector3 = camera.project_ray_origin(pixel)
	var direction: Vector3 = camera.project_ray_normal(pixel)
	var closest: float = INF
	var result: Dictionary = {}
	for mesh: MeshInstance3D in car.find_children("*","MeshInstance3D",true,false):
		var inverse: Transform3D = mesh.global_transform.affine_inverse()
		var local_origin: Vector3 = inverse*origin
		var local_direction: Vector3 = inverse.basis*direction
		var faces: PackedVector3Array = mesh.mesh.get_faces()
		for index: int in range(0,faces.size(),3):
			var point: Variant = Geometry3D.ray_intersects_triangle(local_origin,local_direction,faces[index],faces[index+1],faces[index+2])
			if point != null and origin.distance_squared_to(mesh.global_transform*point)<closest:
				closest = origin.distance_squared_to(mesh.global_transform*point)
				result = {"part":str(mesh.name),"local_hit":str(point),"material":str(mesh.get_active_material(0))}
	return result

func run() -> void:
	var before: bool = "--before" in OS.get_cmdline_user_args()
	for argument: String in OS.get_cmdline_user_args():
		if argument.begins_with("--evidence-root="): output = argument.trim_prefix("--evidence-root=")
	if before:
		var baseline: String = "res://../docs/evidence/electric-quality-pass-2026-10-25/before"
		var author: GDScript = COACH
		author.source_code = FileAccess.get_file_as_string(baseline+"/game/vehicle/electric_coachwork.gd").replace("res://assets/vehicles/electric_coachwork_v7",baseline+"/electric_coachwork_v7")
		assert(author.reload() == OK)
		var assembly: GDScript = MODEL
		assembly.source_code = FileAccess.get_file_as_string(baseline+"/game/vehicle/electric_car_model.gd")
		assert(assembly.reload() == OK)
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(output))
	root.size = Vector2i(1440,900)
	var viewport := SubViewport.new()
	viewport.size = root.size
	viewport.own_world_3d = true
	viewport.msaa_3d = Viewport.MSAA_4X
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(viewport)
	for simple: bool in [false,true]:
		var label: String = "simple" if simple else "full"
		var world := Node3D.new()
		viewport.add_child(world)
		var car := CarVisual.new()
		car.vehicle_id = "electric"
		car.simple_model = simple
		car.batch_parts = false
		car.livery = VehicleLivery.defaults("electric")
		car.livery.colors.body = "263f55"
		car.livery.colors.accent = "d7e0e2"
		world.add_child(car)
		ShowroomStudio.build(world,car)
		var count: int = triangles(car)
		metrics[label+"_triangles"] = count
		check(count < 170000,"unchanged triangle ceiling "+label)
		check(car.body.get_node_or_null("IntegratedTailLip") == null,"retired separate accent-colour tail blade "+label)
		check(car.body.find_children("MirrorShell*","MeshInstance3D",true,false).size() == 2,"two swept mirror shells replace the twelve box faces "+label)
		check(car.body.get_node_or_null("RearLightGuide") != null and car.body.get_node_or_null("RearLens") != null,"light guides sit inside a separate optical lens "+label)
		var centre: Dictionary = mesh_ray(car,"TailSkin",Vector3(0,0.70,3),Vector3.FORWARD)
		var outer: Dictionary = mesh_ray(car,"TailSkin",Vector3(0.70,0.70,3),Vector3.FORWARD)
		var bow: float = centre.point.z-outer.point.z
		metrics[label+"_actual_rear_bow"] = bow
		check(centre.hit and outer.hit and bow>0.06,"actual tail mesh curves across the rear "+label)
		check(car.body.find_children("DiffuserFin*","MeshInstance3D",true,false).is_empty(),"retired dangling rectangular diffuser fins "+label)
		check(car.body.find_children("CabinScreenBase*","MeshInstance3D",true,false).is_empty(),"fitted screen apertures remove buried full paint layers "+label)
		var cavity_facing: bool = true
		for mesh: MeshInstance3D in car.body.find_children("ArchReturn*","MeshInstance3D",true,false):
			var arrays: Array = mesh.mesh.surface_get_arrays(0)
			var vertices: PackedVector3Array = arrays[Mesh.ARRAY_VERTEX]
			for index: int in range(vertices.size()):
				var point: Vector3 = vertices[index]
				var axle: float = -COACH.AXLE if point.z<0 else COACH.AXLE
				var inward: Vector3 = Vector3(0,0.37-point.y,axle-point.z).normalized()
				cavity_facing = cavity_facing and inward.dot(arrays[Mesh.ARRAY_NORMAL][index])>0.98
		check(cavity_facing,"wheel-arch returns face the visible cavity throughout "+label)
		for side: float in [-1.0,1.0]:
			for axle: float in [-COACH.AXLE,COACH.AXLE]:
				for y: float in [0.40,0.55,0.72]:
					var inner: Dictionary = mesh_ray(car,"WheelwellInner",Vector3(side*1.3,y,axle+0.10),Vector3(-side,0,0))
					var expected_x: float=COACH.deform_front_point(Vector3(side*0.48,y,axle+0.10)).x
					check(inner.hit and absf(inner.point.x-expected_x)<0.001,"actual closed wheel tub blocks cross-body sightline "+label+" "+str(side)+"/"+str(axle)+"/"+str(y))
		var camera := Camera3D.new()
		camera.fov = 32
		camera.current = true
		world.add_child(camera)
		var shots: Array = [
			["rear",Vector3(0,1.95,4.7),Vector3(0,0.78,1.25)],
			["rear-quarter",Vector3(-4.9,2.5,6.5),Vector3(0,0.67,0.4)],
			["tail-detail",Vector3(-2.4,1.35,4.2),Vector3(0,0.68,1.9)],
			["front-quarter",Vector3(-5,2.9,-7),Vector3(0,0.65,0)],
			["side",Vector3(-7.5,1.9,0),Vector3(0,0.7,0)],
			["mirror",Vector3(-2.8,1.55,0.8),Vector3(-0.87,1.02,-0.58)],
			["wheel",Vector3(-2.8,0.9,-2.2),Vector3(-0.75,0.37,-1.2)],
			["roof",Vector3(-2.9,4.6,3.7),Vector3(0,0.85,0.3)]]
		for shot: Array in shots:
			camera.position = shot[1]
			camera.look_at(shot[2])
			await settle()
			var pixels: Image = viewport.get_texture().get_image()
			check(pixels.save_png(output.path_join(label+"-"+str(shot[0])+".png")) == OK,"native capture "+label+" "+str(shot[0]))
			if shot[0] == "rear":
				var visible: int = 0
				for sample: int in range(49):
					var x: float = lerpf(-0.75,0.75,sample/48.0)
					var y: float = 0.849+0.007*pow(absf(x)/0.785,4)
					var pixel_position: Vector2 = camera.unproject_position(car.body.to_global(Vector3(x,y,COACH.rear_z(x,y)-0.025)))
					var lit: bool = false
					for dx: int in range(-2,3):
						for dy: int in range(-2,3):
							var colour: Color = pixels.get_pixel(clampi(roundi(pixel_position.x)+dx,0,pixels.get_width()-1),clampi(roundi(pixel_position.y)+dy,0,pixels.get_height()-1))
							lit = lit or (colour.r>0.14 and colour.r>colour.g*1.8 and colour.r>colour.b*1.8)
					if lit: visible += 1
				metrics[label+"_visible_lamp_samples"] = visible
				check(visible == 49,"all 49 native rear guide samples remain visible "+label)
			if shot[0] == "wheel" and "--diagnose" in OS.get_cmdline_user_args():
				for pixel: Vector2 in [Vector2(850,305),Vector2(865,310),Vector2(880,315)]:
					print("ELECTRIC_PICK ",label," ",pixel," ",JSON.stringify(diagnose(car,camera,pixel)))
		car.livery.colors.body = "638d86"
		car.livery.colors.accent = "263c43"
		car.apply_livery(car.livery)
		camera.position = shots[1][1]
		camera.look_at(shots[1][2])
		await settle()
		check(viewport.get_texture().get_image().save_png(output.path_join(label+"-green-rear.png")) == OK,"native green rear capture "+label)
		world.free()
	viewport.free()
	var report := FileAccess.open(output+"/quality.json",FileAccess.WRITE)
	report.store_string(JSON.stringify({"checks":checks,"failures":failures,"metrics":metrics},"\t"))
	report.close()
	print("ELECTRIC_QUALITY: %d checks, %d failures, %s" % [checks,failures,JSON.stringify(metrics)])
	quit(1 if failures else 0)
