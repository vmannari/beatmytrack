## Audit-only observer: retain the existing workload and add arithmetic means.
func quantiles(values: Array[float]) -> Dictionary:
	var result: Dictionary = super.quantiles(values)
	var total: float = 0.0
	for value: float in values: total += value
	result.mean = total / values.size()
	return result
