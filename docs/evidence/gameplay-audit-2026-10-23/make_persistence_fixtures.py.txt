from pathlib import Path
import json, hashlib
out = Path("build/gameplay-audit-2026-10-23")
out.mkdir(parents=True, exist_ok=True)
for count in (0, 1000, 10000):
    records = {hashlib.sha256(("audit-fixture-"+str(i)).encode()).hexdigest(): {"race_ms":100000,"flying_ms":20000} for i in range(count)}
    (out/("records-"+str(count)+".json")).write_text(json.dumps({"schema":2,"records":records,"settings":{}},separators=(",",":")),encoding="utf-8")
