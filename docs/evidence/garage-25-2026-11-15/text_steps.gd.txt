extends SceneTree
func _initialize() -> void:
	for step: float in [0.7,1.2,1.5,2.0,3.0,4.0]:
		var text:=TextMesh.new()
		text.text="BEAT\nMY TRACK"
		text.font=MenuStyle.DISPLAY
		text.font_size=48
		text.pixel_size=0.0047
		text.depth=0.0
		text.curve_step=step
		print("TEXT_STEPS ",step," ",text.get_faces().size()/3)
	quit()
