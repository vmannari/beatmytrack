extends SceneTree
## Cold presentation imports: no Garage or vehicle class references in this probe.
var output: String="res://../build/garage-first-entry-2026-11-14/manual"
var app: Node
var events: Array[Dictionary] = []
var started: int

var release_sent: bool=false

func _initialize() -> void:
	process_frame.connect(func() -> void:
		if not is_instance_valid(app): return
		app._publishing_slots_foreground=true
		if not release_sent and is_instance_valid(app._retained_garage):
			release_sent=true
			var released:=InputEventMouseButton.new()
			released.button_index=MOUSE_BUTTON_LEFT
			app._retained_garage.showroom._input(released))
	run.call_deferred()

func domain_script(name: String) -> Script:
	for declaration: Dictionary in ProjectSettings.get_global_class_list():
		if declaration.class==name: return load(declaration.path)
	return null

func mark(label: String) -> void:
	var row: Dictionary = {"stage":label,"ms":(Time.get_ticks_usec()-started)/1000.0}
	events.append(row)
	print("GARAGE_ENTRY_STAGE ",JSON.stringify(row))

func run() -> void:
	var id: String="compact"
	var fitted: bool=false
	for arg: String in OS.get_cmdline_user_args():
		if arg.begins_with("--output="): output=arg.trim_prefix("--output=")
		if arg.begins_with("--vehicle="): id=arg.trim_prefix("--vehicle=")
		if arg=="--fitted": fitted=true
	output=ProjectSettings.globalize_path(output)
	DirAccess.make_dir_recursive_absolute(output)
	OS.set_environment("BMT_API_URL","")
	root.size=Vector2i(1280,720)
	app=load("res://core/app.tscn").instantiate()
	app.records=domain_script("LocalRecords").new("user://probe/records.json")
	var customisation: Script=domain_script("VehicleCustomisation")
	var parts: Dictionary=customisation.defaults()
	if fitted:
		for category: String in customisation.CATEGORIES: parts[category]=customisation.options(category,id)[1][0]
	app.library=domain_script("TrackLibrary").new("user://probe/tracks")
	app.cloud_root_override="user://probe/cloud"
	root.add_child(app)
	while not app.startup_complete: await process_frame
	# App owns loading its records. Seed the private fixture after that load.
	app.records.settings.selected_car=id
	app.records.settings.vehicle_customisation={}
	app.records.settings.vehicle_customisation[id]=parts
	domain_script("VehicleCatalog").selected=id
	app._garage_appearance_changed()
	app._publishing_slots_foreground=true
	var music: Node=domain_script("GameSoundtrack").instance()
	music.music_enabled=false
	music.menu_player.stop()
	music.menu_player.stream=null
	app._show_screen("tracks")
	for _i: int in range(2): await process_frame
	var imports: Dictionary={}
	for path: String in ["res://ui/garage_page.gd","res://ui/garage_preview_cache.gd","res://vehicle/electric_kits.gd"]: imports[path]=ResourceLoader.has_cached(path)
	started=Time.get_ticks_usec()
	app._show_screen("garage")
	mark("navigation_returned")
	var page: Control=app._retained_garage
	var studio: Node=page.showroom
	var previous: String=""
	var maximum: float=0
	var frames: int=0
	var deadline: int=Time.get_ticks_msec()+120000
	while studio.get("preparing")==true and Time.get_ticks_msec()<deadline:
		var stage: String="waiting"
		if studio.car!=null:
			stage="building_car" if not studio.car.model_ready else "building_room" if not studio.showroom_ready else "warming" if studio.get("_first")!=null else "settling"
		if stage!=previous: mark(stage);previous=stage
		var tick: int=Time.get_ticks_usec()
		await process_frame
		maximum=maxf(maximum,(Time.get_ticks_usec()-tick)/1000.0)
		frames+=1
	mark("live_ready")
	if DisplayServer.get_name()!="headless":
		await RenderingServer.frame_post_draw
		root.get_texture().get_image().save_png(output.path_join("live.png"))
	var report: Dictionary={"events":events,"max_frame_ms":maximum,"frames":frames,"ready":studio.showroom_ready and studio.get("preparing")!=true}
	report.cached_before=imports
	report.vehicle=id
	report.actual_vehicle=studio.car.vehicle_id
	report.fitted=fitted
	report.selection_matches=studio.car.vehicle_id==id and studio.car.customisation==parts
	studio.invalidate_render();studio.reflections.request_draw()
	for _draw: int in range(2):
		await process_frame
		await RenderingServer.frame_post_draw
	report.main_primitives=studio.showroom_viewport.get_render_info(Viewport.RENDER_INFO_TYPE_VISIBLE,Viewport.RENDER_INFO_PRIMITIVES_IN_FRAME)
	report.main_draws=studio.showroom_viewport.get_render_info(Viewport.RENDER_INFO_TYPE_VISIBLE,Viewport.RENDER_INFO_DRAW_CALLS_IN_FRAME)
	report.reflection_primitives=[]
	for capture: SubViewport in studio.reflections.views:
		report.reflection_primitives.append(capture.get_render_info(Viewport.RENDER_INFO_TYPE_VISIBLE,Viewport.RENDER_INFO_PRIMITIVES_IN_FRAME))
	var count: int=0
	for part: MeshInstance3D in studio.car.find_children("*","MeshInstance3D",true,false):
		if part.mesh!=null and part.cast_shadow!=GeometryInstance3D.SHADOW_CASTING_SETTING_SHADOWS_ONLY: count+=part.mesh.get_faces().size()/3
	report.car_triangles=count
	var base: Script=load("res://vehicle/modular_car_base.gd")
	report.base_hits=base.hits
	var holder: Node=load(base.path_for(studio.car)).instantiate()
	var data: Dictionary=holder.get_meta("modular_car_base",{})
	report.base_key_matches=data.get("key",[])==base.key_for(studio.car)
	var validation: Script=load("res://vehicle/prepared_asset_sources.gd")
	var saved: Dictionary=validation.runtime_projection(data.get("sources",{}))
	var current: Dictionary=validation.runtime_hashes(studio.car.model_id)
	report.source_differences=[]
	for path: String in current:
		if current[path]!=saved.get(path): report.source_differences.append(path)
	holder.free()
	app._show_screen("settings")
	started=Time.get_ticks_usec()
	app._show_screen("garage")
	report.reentry_ms=(Time.get_ticks_usec()-started)/1000.0
	report.same_studio=page.showroom==studio
	FileAccess.open(output.path_join("report.json"),FileAccess.WRITE).store_string(JSON.stringify(report,"  "))
	print("GARAGE_FIRST_ENTRY_PROFILE ",JSON.stringify(report))
	app.queue_free()
	for _i: int in range(20): await process_frame
	quit(0 if report.ready and report.same_studio and report.selection_matches else 1)
