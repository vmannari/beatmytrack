extends "res://tests/test_garage_news_native.gd"

func capture(label: String) -> void:
	if label not in ["local-friend_overtake-960","race-world_record-1600"]: return
	print("PROBE ",label)
	var menu: Node = page.tools.find_child("GarageNewsMenu",true,false)
	for title: String in ["Newspaper","NewsScroll","NewsItems","EditorialColumns","LeadStory","NewsHeadline","StoryMedia","NewsPhotograph","NewsCarPhoto","NewsPhotoCaption","NewsCircuit","StoryClosing","NewsArticle","GarageNewsRace","OtherStories"]:
		var item: Control = menu.find_child(title,true,false)
		if item != null:
			print(title," rect=",item.get_rect()," minimum=",item.get_combined_minimum_size())
			if item is Label: print(" lines=",item.get_line_count())
	var map_column: Control = menu.find_child("NewsCircuit",true,false).get_parent()
	for item: Control in map_column.get_children(): print("map row=",item.get_rect()," min=",item.get_combined_minimum_size())
	if label == "race-world_record-1600": quit(0)
