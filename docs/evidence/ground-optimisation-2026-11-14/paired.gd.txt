 var corridor := GDScript.new()
 corridor.source_code = FileAccess.get_file_as_string("res://../build/ground-optimisation-20261114/corridor-before.txt").replace("class_name SplineGateCorridor","")
 assert(corridor.reload() == OK)

 assert(reference.reload() == OK)
 var track = TrackCompiler.compile(JSON.parse_string(FileAccess.get_file_as_string("res://tests/fixtures/star_spline_lap.json")))
 for cycle: int in range(3):
  var rows: Dictionary = {}

   track._race_gates = {}
   load("res://race/checkpoint_layout_cache.gd").trim_to()
