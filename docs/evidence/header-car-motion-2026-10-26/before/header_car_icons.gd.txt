extends Node
## One temporary Profile studio at a time; reopening uses bounded static icons.
const Profile = preload("res://ui/profile_car.gd")
const SIZE := Vector2i(176,112)
const CACHE_LIMIT: int = 14
static var images: Dictionary = {}
var popup: PopupMenu
var studio: Control
var generation: int = 0
var rendered: int = 0
var peak_worlds: int = 0
var active_worlds: int = 0
var _jobs: Array[Dictionary] = []
var _current: Dictionary = {}
var _draws: int = 0

func _ready() -> void:
	set_process(false)
	popup.popup_hide.connect(cancel)
	get_parent().visibility_changed.connect(func() -> void:
		if not get_parent().is_visible_in_tree(): cancel())

func request(row: int, vehicle: String, paint: Dictionary, parts: Dictionary) -> void:
	var key: String = RenderingServer.get_current_rendering_method()+"/"+JSON.stringify([vehicle,paint,parts]).sha256_text()
	if images.has(key):
		popup.set_item_icon(row,images[key])
		return
	if DisplayServer.get_name() == "headless": return
	if _current.get("key","") == key: return
	for job: Dictionary in _jobs:
		if job.key == key: return
	_jobs.append({"row":row,"vehicle":vehicle,"paint":paint.duplicate(true),"parts":parts.duplicate(true),"key":key,"generation":generation})
	set_process(true)
	if not RenderingServer.frame_post_draw.is_connected(_after_draw): RenderingServer.frame_post_draw.connect(_after_draw)

func _valid(job: Dictionary) -> bool:
	return popup.visible and job.generation == generation and job.row < popup.item_count and VehicleCatalog.IDS[popup.get_item_id(job.row)] == job.vehicle

func _process(_delta: float) -> void:
	if studio != null: return
	while not _jobs.is_empty():
		var job: Dictionary = _jobs.pop_front()
		if not _valid(job): continue
		_current = job
		studio = Profile.make_preview(job.vehicle,job.paint,job.parts)
		studio.position = Vector2(-10000,-10000)
		studio.size = Vector2(88,56)
		add_child(studio)
		active_worlds += 1
		studio.tree_exited.connect(func() -> void: active_worlds -= 1,CONNECT_ONE_SHOT)
		Profile.apply_presentation(studio)
		peak_worlds = maxi(peak_worlds,active_worlds)
		_draws = 5
		return
	_stop()

func _after_draw() -> void:
	if studio == null: return
	_draws -= 1
	if _draws == 1:
		studio.viewport.size = SIZE
		studio._fit_camera()
		studio.viewport.render_target_update_mode = SubViewport.UPDATE_ONCE
	elif _draws == 0:
		if _valid(_current):
			var pixels: Image = studio.viewport.get_texture().get_image()
			pixels.generate_mipmaps()
			var texture := ImageTexture.create_from_image(pixels)
			if images.size() >= CACHE_LIMIT: images.erase(images.keys()[0])
			images[_current.key] = texture
			_bind_icon.call_deferred(_current.duplicate(),texture)
			rendered += 1
		_release_studio()

func _bind_icon(job: Dictionary, texture: Texture2D) -> void:
	if not _valid(job): return
	# Update native popup layout outside the renderer's post-draw callback.
	popup.set_item_icon(job.row,texture)

func cancel() -> void:
	generation += 1
	_jobs.clear()
	_release_studio()
	_stop()

func _release_studio() -> void:
	if is_instance_valid(studio):
		remove_child(studio)
		studio.queue_free()
	studio = null
	_current = {}
	_draws = 0

func _stop() -> void:
	set_process(false)
	if RenderingServer.frame_post_draw.is_connected(_after_draw): RenderingServer.frame_post_draw.disconnect(_after_draw)

func _exit_tree() -> void:
	if RenderingServer.frame_post_draw.is_connected(_after_draw): RenderingServer.frame_post_draw.disconnect(_after_draw)
