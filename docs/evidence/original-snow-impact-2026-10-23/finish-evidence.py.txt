"""Retain the exact original snow-impact receipts and a task-only binary patch."""
from pathlib import Path
import hashlib
import json
import re
import shutil
import subprocess

ROOT = Path.cwd().resolve()
OUT = ROOT / "docs/evidence/original-snow-impact-2026-10-23"
BUILD = ROOT / "build/original-snow-impact-2026-10-23"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2) + "\n", encoding="utf8")


logs = OUT / "logs"
logs.mkdir(exist_ok=True)
receipts = {"capture-original-final": "33376", "capture-initial-redundant-checks": "24032", "driving-capture": "37152", "snowbank-headless": "29876", "response-headless": "13980", "response-compatibility": "12596", "windows-impacts": "25736", "response-desktop-mobile": "28080", "parser": "6120"}
results = []
for label, identity in receipts.items():
    source = ROOT / f"build/logs/TestScript-{identity}/test-script.log"
    shutil.copyfile(source, logs / (label + ".log"))
    build = source.parent / "client-build-identity.json"
    if build.exists(): shutil.copyfile(build, logs / (label + "-build.json"))
    results.append(dict(label=label, source=str(source.relative_to(ROOT)), summary=[line for line in source.read_text().splitlines() if re.search(r"checks,|SCRIPT_PARSE:|FAIL:|ERROR:", line)]))
shutil.copyfile(BUILD / "import.log", logs / "import.log")
variants = OUT / "variants"
variants.mkdir(exist_ok=True)
for source in (BUILD / "variants").iterdir():
    if source.is_file(): shutil.copyfile(source, variants / source.name)
analysis = json.loads((OUT / "analysis.json").read_text())
assert analysis["failures"] == 0 and len(analysis["checks"]) == 34
for unchanged in ["game/vehicle/vehicle_audio.gd", "game/assets/audio/vehicles/snow.wav"]:
    assert (ROOT / unchanged).read_bytes() == (OUT / "before" / unchanged).read_bytes(), unchanged
save("report.json", dict(schema="bmt-original-snow-impact-v1", date="2026-10-23",
    requirements={"OSI01": "Author three original cleared-source snow impacts", "OSI02": "Install a stronger snow entry while retaining prompt attack, headroom and exact driving", "OSI03": "Provide equal-gain/equal-pitch actual event and bank-driving auditions", "R01": "Native, lifecycle, source and strength checks", "Q01": "Source provenance and scoped Git-checkable checkpoint"},
    engine="4.7.2.stable.official.ed1daf0bf", installed_variant="packed", analysis="analysis.json",
    stronger_mastered_entry_db=analysis["native_master_captures"]["packed"]["early_100ms_dbfs"] - analysis["native_master_captures"]["before"]["early_100ms_dbfs"],
    samples=analysis["samples"], native_captures=analysis["native_master_captures"], receipts=results,
    commands=["python scripts/build_snow_audio.py", "Repeat Python build with before/after installed WAV/manifest SHA256 assertion: exact", "powershell -NoProfile -ExecutionPolicy Bypass -File scripts/dev.ps1 -Action Import -RenderingMethod gl_compatibility", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/capture_original_snow_impact.gd -Graphical -RenderingMethod gl_compatibility", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/test_snow_bank_audio.gd", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/test_snow_audio_response.gd (headless; -Graphical -RenderingMethod gl_compatibility; -Graphical -RenderingMethod mobile, separately)", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/test_windows_impact_audio.gd -Graphical -RenderingMethod gl_compatibility", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/capture_snow_bank_audio.gd -Graphical -RenderingMethod gl_compatibility -TestArgument --output-dir=res://../docs/evidence/original-snow-impact-2026-10-23/driving", "python scripts/review_original_snow_impact.py", "scripts/dev.ps1 -Action TestScript -ScriptPath res://tests/test_script_parse.gd", "Independent Git index diff --binary / whitespace / reverse apply checks; shared index unchanged"],
    reference=dict(url="https://www.soundsnap.com/search/audio?query=snow+falling+heavy", usable_preview=False, auditory_comparison_claimed=False, source_audio_used=False, approach="Owner-authorized original composites from existing Nox CC0 snow recordings"),
    sources="game/assets/audio/vehicles/snow_manifest.json", reproduction="Default packed build repeats installed WAV/manifest hashes exactly; source hashes verified",
    limits=["Owner sound-character and physical-device listening remain", "Restart client to load cached sample replacement", "No phone/tablet attachment check, physical-device test, APK/export/install, native rebuild, account/backend mutation, purchase or deployment", "No owner editor/game was terminated", "No Soundsnap recording used or claimed as listened to", "All auditions use the same gain/pitch/mastering; actual cue direction and output paths remain unchanged", "Concurrent work preserved; no shared Git commit"],
    corrections=["Initial 17-check capture contained four redundant fresh-state comparisons. Removed them and reran the 13 meaningful native capture checks successfully."]))

# Separate staging/index supports an exact binary task patch without committing
# unrelated live changes or touching the shared repository's index.
baseline = {}
for directory in ["before", "doc-base"]:
    for path in (OUT / directory).rglob("*"):
        if path.is_file(): baseline[str(path.relative_to(OUT / directory)).replace("\\", "/")] = path
new = ["game/tests/capture_original_snow_impact.gd", "game/tests/capture_original_snow_impact.gd.uid", "scripts/review_original_snow_impact.py", "docs/ORIGINAL_SNOW_IMPACT_2026_10_23.md"]
# Auditions/fixtures/reports are part of the reviewable checkpoint. Staging bases
# and the checkpoint itself remain evidence inputs, avoiding recursive patches.
for path in OUT.rglob("*"):
    if path.is_file() and not any(part in {"doc-base"} for part in path.relative_to(OUT).parts) and path.name not in {"workspace-task-only.patch", "checkpoint.json", "finish-evidence.py.txt"}:
        new.append(str(path.relative_to(ROOT)).replace("\\", "/"))
stage = BUILD / "checkpoint"
assert stage.resolve().is_relative_to(BUILD.resolve())
stage.mkdir(exist_ok=True)


def git(*args, output=False):
    return subprocess.run(["git", "-C", str(stage), *args], check=True, stdout=subprocess.PIPE if output else subprocess.DEVNULL).stdout


git("init", "-q")
git("config", "core.autocrlf", "false")
git("config", "core.whitespace", "blank-at-eol,blank-at-eof,space-before-tab,cr-at-eol")
for name, source in baseline.items():
    destination = stage / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)
git("add", "--", *baseline)
manifest = []
for name in list(dict.fromkeys([*baseline, *new])):
    source = ROOT / name
    destination = stage / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)
    old = baseline.get(name)
    if old is None or old.read_bytes() != source.read_bytes():
        manifest.append(dict(path=name, before_sha256=sha(old) if old else None, sha256=sha(source), bytes=source.stat().st_size))
git("add", "-N", "--", *new)
patch = OUT / "workspace-task-only.patch"
patch.write_bytes(git("diff", "--binary", "--no-ext-diff", output=True))
git("diff", "--check")
git("apply", "--reverse", "--check", str(patch))
subprocess.run(["git", "apply", "--reverse", "--check", str(patch)], cwd=ROOT, check=True)
save("checkpoint.json", dict(schema="bmt-workspace-checkpoint-v1", baseline="Saved task-start sources and current shared-document bases; independent Git index", files=manifest, patch_sha256=sha(patch), shared_index_modified=False, git_commit_created=False, whitespace="pass", staged_reverse_check="pass", shared_workspace_reverse_check="pass"))
shutil.copyfile(Path(__file__), OUT / "finish-evidence.py.txt")
print("ORIGINAL_SNOW_CHECKPOINT:", len(manifest), "files, binary patch / source hashes / reverse checks pass")
