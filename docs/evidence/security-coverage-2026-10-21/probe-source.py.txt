"""Disposable probes for the current application; no live service requests."""
import importlib.util
from pathlib import Path
import re
from uuid import uuid4

from fastapi.routing import iter_route_contexts

root = Path(__file__).resolve().parents[2]
spec = importlib.util.spec_from_file_location('previous_security_probes', root / 'docs/evidence/backend-attack-review-2026-10-14/test_attack_probes.py')
previous = importlib.util.module_from_spec(spec)
spec.loader.exec_module(previous)
audit = previous.audit
test_cross_player_private_access_is_denied = previous.test_cross_player_private_access_is_denied
test_manufactured_plausible_trajectory_is_accepted_by_design = previous.test_manufactured_plausible_trajectory_is_accepted_by_design


def test_all_current_admin_get_routes_deny_game_credentials(audit):
    checked = []
    for context in iter_route_contexts(audit.app.router.routes):
        if not context.path.startswith('/v1/admin/') or 'GET' not in context.original_route.methods:
            continue
        target = re.sub(r'\{[^}]+\}', str(uuid4()), context.path)
        response = audit.client.get(target, headers=audit.auth)
        assert response.status_code == 401, (context.path, response.status_code, response.text)
        checked.append(context.path)
    assert checked
    (Path(__file__).parent / 'admin-routes-checked.json').write_text(__import__('json').dumps(checked, indent=2))
