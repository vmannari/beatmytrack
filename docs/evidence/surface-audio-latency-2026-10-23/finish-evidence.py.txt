def git(*args, output=False):
    return subprocess.run(["git", "-C", str(stage), *args], check=True, stdout=subprocess.PIPE if output else subprocess.DEVNULL).stdout


git("init", "-q")
git("config", "core.autocrlf", "false")
git("config", "core.whitespace", "blank-at-eol,blank-at-eof,space-before-tab,cr-at-eol")
for name, before in current.items():
    target = stage / name
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(before, target)

manifest = []
for name in [*current, *new]:
    target = stage / name
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(ROOT / name, target)
    before = current.get(name)
    if not before or before.read_bytes() != target.read_bytes():
        manifest.append(dict(path=name, before_sha256=sha(before) if before else None, after_sha256=sha(target), bytes=target.stat().st_size))
