shader_type spatial;
render_mode diffuse_burley, cull_back;
varying vec3 craft_position;
void vertex() { craft_position=(MODEL_MATRIX*vec4(VERTEX,1.0)).xyz; }
void fragment() {
	vec3 pigment=COLOR.rgb;
	if (!OUTPUT_IS_SRGB) pigment=mix(pigment/12.92,pow((pigment+0.055)/1.055,vec3(2.4)),step(vec3(0.04045),pigment));
	float filter_width=length(fwidth(craft_position));
	float fine=1.0-smoothstep(0.008,0.05,filter_width);
	float grain=fract(sin(dot(floor(craft_position*160.0),vec3(19.1,73.7,31.3)))*43758.5453);
	ALBEDO=pigment*(1.0+(grain-0.5)*0.045*fine);
	ROUGHNESS=0.91; SPECULAR=0.20;
	if (COLOR.a>0.9) { EMISSION=pigment*2.2; ROUGHNESS=1.0; }
}
